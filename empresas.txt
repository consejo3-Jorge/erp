  `espacio_plazo1` smallint(6) DEFAULT 0, -- en caso de que el cliente nos pague mediante giro bancario, indica el número de días a dejar entre la factura y el primer giro bancario que se le emite al cliente

  `costes_directos` decimal(10,2) DEFAULT 0.00,  --se desconoce su uso
  `costes_indirectos` decimal(10,2) DEFAULT 0.00, --se desconoce su uso
  `costes_salariales` decimal(10,2) DEFAULT 0.00, --se desconoce su uso
  `no_envio_email_creditos` smallint(6) DEFAULT 0, --indica si para este cliente, debemos evitar enviarle información de los créditos disponibles que tiene (1=No se envían emails de créditos, 0=si se envían emails de créditos)

  `ver_tiempos_inf_segui` smallint(6) DEFAULT 0, -- indica si este cliente, cuando usa el área de clientes y solicita ver un informe de seguimiento, si ve los tiempos reales empleados por los alumno o no. (1=Si ve los tiempos realies, 0=no ve los tiempos reales)

  `url_panel_control` varchar(255) DEFAULT NULL, -- contiene la URL que nos dirige a cargar un panel de control personalizado para el cliente
  `id_clasif` smallint(6) DEFAULT 0, -- contiene el ID de una clasificacióno del cliente en función de su actividad comercial. Se relaciona con el campo erp_empresas_clasificacion.id
  `automatricular` smallint(6) DEFAULT NULL, -- indica si el cliente está autorizado a automatricular a sus alumnos usando nuestro servicio área de clientes. (1=autorizado, 0=no autorizado)
